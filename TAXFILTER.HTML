Sub tax_CLEAN() ' SPR + PRD + MATCH - Clean Version - Har Gaon Pe Chalega ' Keyboard Shortcut: Ctrl + t Application.ScreenUpdating = False Dim wsSPR As Worksheet, wsPRD As Worksheet Dim ws1 As Worksheet, ws2 As Worksheet, ws3 As Worksheet, ws4 As Worksheet Dim lastRow As Long, i As Long Dim dict As Object, key As String Dim arr As Variant Set dict = CreateObject("Scripting.Dictionary") ' ========== 1. SPR - Family ID Duplicate Hatao ========== Set wsSPR = ThisWorkbook.Sheets("spr") ' Agar SPR sheet ka naam alag hai to yaha badlo lastRow = wsSPR.Cells(wsSPR.Rows.Count, "B").End(xlUp).Row ' B = Family ID Set ws1 = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count)) ws1.Name = "total spr families" ws1.Range("A1").Value = "S.No." ws1.Range("B1").Value = "Family ID" ' Family ID ka pehla wala hi rakho - Duplicate apne aap hat jayega Dim r As Long, outRow As Long outRow = 2 For r = 2 To lastRow key = Trim(CStr(wsSPR.Cells(r, "B").Value)) If key <> "" And key <> "0" And Not dict.Exists(key) Then dict.Add key, 1 ws1.Cells(outRow, 1).Value = outRow - 1 ws1.Cells(outRow, 2).Value = key outRow = outRow + 1 End If Next r MsgBox "STEP 1 Ho Gaya! SPR Final Family: " & dict.Count ' ========== 2. PRD - Text To Column + Duplicate + Unique ========== Set wsPRD = ThisWorkbook.Sheets("prd") ' Agar PRD sheet ka naam alag hai to yaha badlo ' B Column pe Text To Column - Fixed Width - Aapke purane wale hisab se wsPRD.Columns("B:B").TextToColumns Destination:=Range("B1"), DataType:=xlFixedWidth, _ FieldInfo:=Array(Array(0, 1), Array(6, 1), Array(9, 1), Array(33, 1), Array(65, 1), _ Array(154, 1), Array(159, 1), Array(164, 1), Array(166, 1), Array(264, 1)), _ TrailingMinusNumbers:=True lastRow = wsPRD.Cells(wsPRD.Rows.Count, "B").End(xlUp).Row ' Family ID kis Column me aayi? - TextToColumn ke baad B me hi hogi ' Duplicate Count karo Set dict = CreateObject("Scripting.Dictionary") Dim dictCount As Object Set dictCount = CreateObject("Scripting.Dictionary") For r = 2 To lastRow ' B Column me se sirf number nikalo - "Family ID : 233..." me se 233 nikalo Dim txt As String, fid As String txt = Trim(CStr(wsPRD.Cells(r, "B").Value)) ' Agar B me "Family ID" likha hai to uske baad wala number lo If InStr(txt, "Family") > 0 Then fid = Trim(Replace(Replace(txt, "Family ID", ""), ":", "")) Else fid = txt End If ' Sirf number rakho fid = Trim(fid) If fid <> "" And fid <> "0" Then If dictCount.Exists(fid) Then dictCount(fid) = dictCount(fid) + 1 Else dictCount.Add fid, 1 End If Next r ' Duplicate wali sheet Set ws3 = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count)) ws3.Name = "duplicate in prd" ws3.Range("A1").Value = "Family ID" ws3.Range("B1").Value = "Kitni Baar Aayi" outRow = 2 For Each key In dictCount.Keys If dictCount(key) > 1 Then ws3.Cells(outRow, 1).Value = key ws3.Cells(outRow, 2).Value = dictCount(key) outRow = outRow + 1 End If Next key ' Unique wali sheet Set ws2 = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count)) ws2.Name = "total prd unique families" ws2.Range("A1").Value = "S.No." ws2.Range("B1").Value = "Family ID" outRow = 2 For Each key In dictCount.Keys ws2.Cells(outRow, 1).Value = outRow - 1 ws2.Cells(outRow, 2).Value = key outRow = outRow + 1 Next key MsgBox "STEP 2 Ho Gaya! PRD Unique: " & dictCount.Count & " | Duplicate Family: " & (ws3.Cells(ws3.Rows.Count, "A").End(xlUp).Row - 1) ' ========== 3. MATCH - SPR vs PRD Unique ========== ' PRD Unique ko Dictionary me dalo Set dict = CreateObject("Scripting.Dictionary") Dim prdLast As Long prdLast = ws2.Cells(ws2.Rows.Count, "B").End(xlUp).Row For r = 2 To prdLast key = Trim(CStr(ws2.Cells(r, "B").Value)) If key <> "" Then dict.Add key, 1 Next r ' Remaining - Jo SPR me hai par PRD me nahi hai Set ws4 = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count)) ws4.Name = "remaining for tax registration" ws4.Range("A1").Value = "S.No." ws4.Range("B1").Value = "Family ID" outRow = 2 Dim sprLast As Long sprLast = ws1.Cells(ws1.Rows.Count, "B").End(xlUp).Row For r = 2 To sprLast key = Trim(CStr(ws1.Cells(r, "B").Value)) If Not dict.Exists(key) Then ws4.Cells(outRow, 1).Value = outRow - 1 ws4.Cells(outRow, 2).Value = key outRow = outRow + 1 End If Next r Application.ScreenUpdating = True ThisWorkbook.Save MsgBox "FINAL HO GAYA! Remaining SPR: " & (outRow - 2) & " | 4 Sheet Ban Gayi - total spr families, duplicate in prd, total prd unique families, remaining for tax registration" End Sub
