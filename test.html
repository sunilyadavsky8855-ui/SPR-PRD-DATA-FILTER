const SHEETS = { Registration: ["Date","Receipt","Village","Groom","Bride","Father","Mobile","Amount","Operator"], Donations: ["Date","Receipt","Village","Name","Father","Amount","Type","Operator"], Expenses: ["Date","Receipt","Item","Description","PaidBy","Amount","Operator"], Users: ["Username","PasswordHash","Role","Name"] }; function doGet() { setupSheets(); return HtmlService.createHtmlOutputFromFile("Index") .setTitle("सम्मेलन मैनेजमेंट सिस्टम") .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } function setupSheets() { const ss = SpreadsheetApp.getActiveSpreadsheet(); Object.keys(SHEETS).forEach(name => { let sh = ss.getSheetByName(name); if (!sh) sh = ss.insertSheet(name); if (sh.getLastRow() === 0) { sh.appendRow(SHEETS[name]); } }); const users = ss.getSheetByName("Users"); if (users.getLastRow() < 2) { users.appendRow([ "admin", hashPassword("Admin@123"), "admin", "मुख्य व्यवस्थापक" ]); } } function hashPassword(password) { const bytes = Utilities.computeDigest( Utilities.DigestAlgorithm.SHA_256, password, Utilities.Charset.UTF_8 ); return bytes.map(b => ("0" + ((b + 256) % 256).toString(16)).slice(-2)).join(""); } function login(username, password) { const sh = SpreadsheetApp.getActive().getSheetByName("Users"); const data = sh.getDataRange().getValues(); const user = data.slice(1).find(r => String(r[0]) === String(username) && String(r[1]) === hashPassword(password) ); if (!user) return {success:false, message:"गलत यूजर आईडी या पासवर्ड"}; const token = Utilities.getUuid(); CacheService.getScriptCache().put( "session_" + token, JSON.stringify({ username:user[0], role:user[2], name:user[3] }), 21600 ); return { success:true, token, role:user[2], name:user[3] }; } function getUser(token) { const value = CacheService.getScriptCache() .get("session_" + token); if (!value) throw new Error("Session समाप्त। दोबारा लॉगिन करें।"); return JSON.parse(value); } function nextReceipt(sheetName) { const sh = SpreadsheetApp.getActive().getSheetByName(sheetName); return sheetName.substring(0,1).toUpperCase() + Utilities.formatString("%05d", sh.getLastRow()); } function saveRegistration(token, d) { const u = getUser(token); const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const sh = SpreadsheetApp.getActive().getSheetByName("Registration"); const receipt = nextReceipt("Registration"); sh.appendRow([ new Date(), receipt, d.village, d.groom, d.bride, d.father, d.mobile, Number(d.amount || 0), u.username ]); return {success:true, receipt}; } finally { lock.releaseLock(); } } function saveDonation(token, d) { const u = getUser(token); const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const sh = SpreadsheetApp.getActive().getSheetByName("Donations"); const receipt = nextReceipt("Donations"); sh.appendRow([ new Date(), receipt, d.village, d.type === "गुप्त" ? "गुप्त दान" : d.name, d.type === "गुप्त" ? "" : d.father, Number(d.amount), d.type, u.username ]); return {success:true, receipt}; } finally { lock.releaseLock(); } } function saveExpense(token, d) { const u = getUser(token); const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const sh = SpreadsheetApp.getActive().getSheetByName("Expenses"); const receipt = nextReceipt("Expenses"); sh.appendRow([ new Date(), receipt, d.item, d.description, d.paidBy, Number(d.amount), u.username ]); return {success:true, receipt}; } finally { lock.releaseLock(); } } function getDashboard(token) { const u = getUser(token); const ss = SpreadsheetApp.getActive(); const read = name => ss.getSheetByName(name) .getDataRange().getValues().slice(1); let registrations = read("Registration"); let donations = read("Donations"); let expenses = read("Expenses"); if (u.role !== "admin") { registrations = registrations.filter(r => r[8] === u.username); donations = donations.filter(r => r[7] === u.username); expenses = expenses.filter(r => r[6] === u.username); } const total = (rows, index) => rows.reduce((s,r) => s + Number(r[index] || 0), 0); const villages = {}; registrations.forEach(r => { const v = r[2] || "अन्य"; villages[v] = (villages[v] || 0) + 1; }); return { role:u.role, registrations:registrations.length, donation:total(donations,5), expenses:total(expenses,5), villages, balance:total(registrations,7) + total(donations,5) - total(expenses,5) }; } function addOperator(token, username, password, name) { const u = getUser(token); if (u.role !== "admin") throw new Error("केवल Admin यह कर सकता है"); if (!username || !password) throw new Error("सभी जानकारी भरें"); const sh = SpreadsheetApp.getActive().getSheetByName("Users"); const data = sh.getDataRange().getValues(); if (data.slice(1).some(r => r[0] === username)) { throw new Error("यह यूजर पहले से मौजूद है"); } sh.appendRow([ username, hashPassword(password), "operator", name ]); return "ऑपरेटर सफलतापूर्वक बनाया गया"; }
