const SHEET_LINK = "https://docs.google.com/spreadsheets/d/1gixoGfKTs3dr0gelDGfQ23oKbwyUgw7mfgU_gsPjMXY/edit"; function getSS(){ return SpreadsheetApp.openByUrl(SHEET_LINK); } const SHEETS = { Registration: ["Date","Receipt","Village","Groom","Bride","Father","Mobile","Amount","Operator"], Donations: ["Date","Receipt","Village","Name","Father","Amount","Type","Operator"], Expenses: ["Date","Receipt","Item","Description","PaidBy","Amount","Operator"], Users: ["Username","PasswordHash","Role","Name"] }; function setupSheets() { const ss = getSS(); Object.keys(SHEETS).forEach(name => { let sh = ss.getSheetByName(name); if (!sh) sh = ss.insertSheet(name); if (sh.getLastRow() === 0) sh.appendRow(SHEETS[name]); }); const users = ss.getSheetByName("Users"); if (users.getLastRow() < 2) users.appendRow(["admin", hashPassword("Admin@123"), "admin", "मुख्य व्यवस्थापक"]); } function hashPassword(p){ const b=Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256,p,Utilities.Charset.UTF_8); return b.map(x=>("0"+((x+256)%256).toString(16)).slice(-2)).join(""); } function getUserByToken(t){ const v=CacheService.getScriptCache().get("session_"+t); return v?JSON.parse(v):null; } function doPost(e){ setupSheets(); const d=JSON.parse(e.postData.contents); const out=ContentService.createTextOutput(); out.setMimeType(ContentService.MimeType.JSON); try{ if(d.action==="login"){ const sh=getSS().getSheetByName("Users"); const data=sh.getDataRange().getValues(); const user=data.slice(1).find(r=>String(r[0])===String(d.username)&&String(r[1])===hashPassword(d.password)); if(!user) return out.setContent(JSON.stringify({success:false,message:"गलत यूजर आईडी या पासवर्ड"})); const token=Utilities.getUuid(); CacheService.getScriptCache().put("session_"+token,JSON.stringify({username:user[0],role:user[2],name:user[3]}),21600); return out.setContent(JSON.stringify({success:true,token,role:user[2],name:user[3]})); } if(d.action==="dashboard"){ const u=getUserByToken(d.token); if(!u) return out.setContent(JSON.stringify({success:false,message:"Session समाप्त"})); const ss=getSS(); const read=n=>ss.getSheetByName(n).getDataRange().getValues().slice(1); let regs=read("Registration"),dons=read("Donations"),exps=read("Expenses"); if(u.role!=="admin"){ regs=regs.filter(r=>r[8]===u.username); dons=dons.filter(r=>r[7]===u.username); exps=exps.filter(r=>r[6]===u.username); } const total=(rows,idx)=>rows.reduce((s,r)=>s+Number(r[idx]||0),0); const villages={}; regs.forEach(r=>{const v=r[2]||"अन्य"; villages[v]=(villages[v]||0)+1;}); return out.setContent(JSON.stringify({success:true,registrations:regs.length,donation:total(dons,5),expenses:total(exps,5),villages,balance:total(regs,7)+total(dons,5)-total(exps,5)})); } if(["saveReg","saveDon","saveExp","addOp"].includes(d.action)){ const u=getUserByToken(d.token); if(!u) return out.setContent(JSON.stringify({success:false,message:"Session समाप्त"})); const ss=getSS(); const lock=LockService.getScriptLock(); lock.waitLock(10000); try{ if(d.action==="saveReg"){ const sh=ss.getSheetByName("Registration"); const receipt="R"+Utilities.formatString("%05d",sh.getLastRow()); sh.appendRow([new Date(),receipt,d.data.village,d.data.groom,d.data.bride,d.data.father,d.data.mobile,Number(d.data.amount||0),u.username]); return out.setContent(JSON.stringify({success:true,receipt})); } if(d.action==="saveDon"){ const sh=ss.getSheetByName("Donations"); const receipt="D"+Utilities.formatString("%05d",sh.getLastRow()); const isGupt=d.data.type==="गुप्त"; sh.appendRow([new Date(),receipt,d.data.village,isGupt?"गुप्त दान":d.data.name,isGupt?"":d.data.father,Number(d.data.amount),d.data.type,u.username]); return out.setContent(JSON.stringify({success:true,receipt})); } if(d.action==="saveExp"){ const sh=ss.getSheetByName("Expenses"); const receipt="E"+Utilities.formatString("%05d",sh.getLastRow()); sh.appendRow([new Date(),receipt,d.data.item,d.data.description,d.data.paidBy,Number(d.data.amount),u.username]); return out.setContent(JSON.stringify({success:true,receipt})); } if(d.action==="addOp"){ if(u.role!=="admin") return out.setContent(JSON.stringify({success:false,message:"केवल Admin"})); const sh=ss.getSheetByName("Users"); const data=sh.getDataRange().getValues(); if(data.slice(1).some(r=>r[0]===d.data.username)) return out.setContent(JSON.stringify({success:false,message:"यूजर पहले से है"})); sh.appendRow([d.data.username,hashPassword(d.data.password),"operator",d.data.name]); return out.setContent(JSON.stringify({success:true,message:"ऑपरेटर बन गया"})); } } finally{ lock.releaseLock(); } } return out.setContent(JSON.stringify({success:false,message:"Unknown"})); }catch(err){ return out.setContent(JSON.stringify({success:false,message:err.message})); } } function doGet(e){ setupSheets(); return ContentService.createTextOutput(JSON.stringify({status:"OK - Sammelan API Chal Rahi Hai"})).setMimeType(ContentService.MimeType.JSON); }
