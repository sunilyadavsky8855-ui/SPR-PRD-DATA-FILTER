function doGet() { return HtmlService.createHtmlOutputFromFile('Index') .setTitle('सम्मेलन प्रबंधन') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.DEFAULT); } function setupSheets() { const ss = SpreadsheetApp.getActiveSpreadsheet(); Object.keys(SHEETS).forEach(key => { const config = SHEETS[key]; let sheet = ss.getSheetByName(config.name); if (!sheet) sheet = ss.insertSheet(config.name); if (sheet.getLastRow() === 0) { sheet.getRange(1, 1, 1, config.headers.length) .setValues([config.headers]); sheet.setFrozenRows(1); } }); return 'आवश्यक शीट तैयार हो गई हैं।'; } function getPublicDashboard() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const registrations = readRows_(ss, 'Registration'); const donations = readRows_(ss, 'Donations'); const expenses = readRows_(ss, 'Expenses'); const registrationByVillage = {}; registrations.forEach(r => { const village = String(r.Village || 'गाँव दर्ज नहीं').trim(); registrationByVillage[village] = (registrationByVillage[village] || 0) + 1; }); const donationByVillage = {}; let regularTotal = 0; let secretTotal = 0; donations.forEach(r => { const village = String(r.Village || 'गाँव दर्ज नहीं').trim(); const amount = Number(r.Amount) || 0; const type = String(r.DonationType || '').trim().toLowerCase(); if (!donationByVillage[village]) { donationByVillage[village] = { regular: 0, secret: 0, total: 0 }; } if (type === 'secret' || type === 'गुप्त') { secretTotal += amount; donationByVillage[village].secret += amount; } else { regularTotal += amount; donationByVillage[village].regular += amount; } donationByVillage[village].total += amount; }); const expenseTotal = expenses.reduce((sum, r) => { return sum + (Number(r.Amount) || 0); }, 0); const villages = [...new Set([ ...Object.keys(registrationByVillage), ...Object.keys(donationByVillage) ])].sort(); const villageRows = villages.map(village => ({ village: village, registrations: registrationByVillage[village] || 0, regular: donationByVillage[village]?.regular || 0, secret: donationByVillage[village]?.secret || 0, total: donationByVillage[village]?.total || 0 })); const donationTotal = regularTotal + secretTotal; return { registrationCount: registrations.length, regularTotal: regularTotal, secretTotal: secretTotal, donationTotal: donationTotal, expenseTotal: expenseTotal, balance: donationTotal - expenseTotal, villageRows: villageRows }; } function readRows_(ss, sheetName) { const sheet = ss.getSheetByName(sheetName); if (!sheet || sheet.getLastRow() < 2) return []; const values = sheet.getDataRange().getValues(); const headers = values.shift().map(h => String(h).trim()); return values .filter(row => row.some(cell => cell !== '' && cell !== null)) .map(row => { const obj = {}; headers.forEach((header, i) => obj[header] = row[i]); return obj; }); } ---------------------------------------- भाग 2 — Index.html ---------------------------------------- सम्मेलन प्रबंधन
